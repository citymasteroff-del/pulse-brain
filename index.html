Google Таблица — по шагам

  1. Создай новую Гугл Таблицу (не смешивай с CRM-заявками)
  2. Первая строка: id, date, source, name, text, outcome, script, speed, aiVerdict, notes
  3. Расширения → Apps Script → удали всё, вставь код ниже → 💾
  4. Развернуть → Новое развёртывание → «Веб-приложение» → доступ: «Все» → Развернуть
  5. Скопируй URL …/exec, вставь ниже, жми «Проверить связь»
  6. Точка «Таблица» в сайдбаре зазеленела — порядок
🔒 Этот адрес — ключ от всей базы. Не пересылай его в чатах и не показывай на скриншотах: у кого он есть, тот скачает все переписки и телефоны.
Таблица — единая копия приложения: выгрузка полностью заменяет лист, поэтому удалённые записи исчезают и не возвращаются. Для этого нужен код Apps Script v5 (ниже) и новое развёртывание после его замены.

Claude API (не обязательно)

Внутри claude.ai ключ не нужен — вызовы бесплатные. Для локального файла: console.anthropic.com → API Keys. Пока платёжка капризничает — живи на Мозге, он не обидится.

Код Apps Script для листа «Диалоги»

Фикс #ERROR! — если в столбцах появились красные #ERROR!, это Гугл принял текст, начинающийся с «+7…» или «=», за формулу. Код ниже уже лечит: пишет ячейки как простой текст. Что сделать: 1) замени код в Apps Script на этот, 2) Развернуть → Управление развёртываниями → карандаш → Версия: Новая → Развернуть (без этого правки не применятся!), 3) выдели битые столбцы в таблице → Формат → Числа → Текст, 4) в файле жми «⬆ Выгрузить базу» — строки перезапишутся начисто.
Зачем ключ: без него любой, кто узнает адрес /exec, скачает всю базу — телефоны и переписки клиентов. Ключ превращает адрес в замок: без него сервер отвечает «Доступ запрещён». Придумай свой (буквы и цифры, 20+ символов) и впиши его в строку SECRET ниже.
// v5: полная синхронизация (замена листа) + ключ + защита от #ERROR!

// ▼▼▼ ЗАМЕНИ НА СВОЙ КЛЮЧ ▼▼▼
var SECRET = 'gani-miuejm7udans90pyrpzy';
// ▲▲▲ он же идёт в конец URL: ...exec?key=ТВОЙ_КЛЮЧ ▲▲▲

var HEADERS = ['id','date','source','name','text','outcome',
               'script','speed','aiVerdict','notes'];

function checkKey_(e) {
  var given = (e && e.parameter) ? e.parameter.key : '';
  if (given !== SECRET) throw new Error('Доступ запрещён: неверный ключ');
}
function deny_(msg) {
  return ContentService.createTextOutput(JSON.stringify({ error: msg }))
    .setMimeType(ContentService.MimeType.JSON);
}
function ok_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
function safeCell_(v) {
  var t = (v === null || v === undefined) ? '' : String(v);
  return /^[=+\-@]/.test(t) ? "'" + t : t;   // не даём Гуглу принять текст за формулу
}

function doPost(e) {
  try { checkKey_(e); } catch (err) { return deny_(err.message); }
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var body = JSON.parse(e.postData.contents);
  var items = Array.isArray(body) ? body : (body.rows || [body]);
  var replace = !Array.isArray(body) && body.mode === 'replace';

  if (replace) {
    // ПОЛНАЯ ЗАМЕНА: лист становится точной копией приложения,
    // удалённые записи исчезают и не возвращаются при следующей загрузке
    sheet.clear();
    var out = [HEADERS];
    items.forEach(function(d) {
      out.push(HEADERS.map(function(h) { return safeCell_(d[h]); }));
    });
    var range = sheet.getRange(1, 1, out.length, HEADERS.length);
    range.setNumberFormat('@');
    range.setValues(out);
    return ok_({ ok: true, mode: 'replace', rows: items.length });
  }

  // обычный режим: дописываем или обновляем по id
  var rows = sheet.getDataRange().getValues();
  items.forEach(function(d) {
    var rowIndex = -1;
    for (var i = 1; i < rows.length; i++) {
      if (rows[i][0] == d.id) { rowIndex = i + 1; break; }
    }
    var r = rowIndex > 0 ? rowIndex : sheet.getLastRow() + 1;
    var range = sheet.getRange(r, 1, 1, HEADERS.length);
    range.setNumberFormat('@');
    range.setValues([HEADERS.map(function(h) { return safeCell_(d[h]); })]);
    if (rowIndex < 0) rows.push([d.id]);
  });
  return ok_({ ok: true, rows: items.length });
}

function doGet(e) {
  try { checkKey_(e); } catch (err) { return deny_(err.message); }
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  var out = [];
  for (var i = 1; i < rows.length; i++) {
    var r = rows[i];
    if (!r[0]) continue;
    var o = {};
    for (var j = 0; j < HEADERS.length; j++) o[HEADERS[j]] = r[j];
    out.push(o);
  }
  return ok_(out);
}
После замены кода обязательно: Развернуть → Управление развёртываниями → карандаш → Версия: Новая → Развернуть. Без этого Гугл продолжит крутить старый код.
